  ux[which.max(tabulate(match(x, ux)))]

prices <- df$PESO

# Rango
R <- max(prices, na.rm = TRUE) - min(prices, na.rm = TRUE)
cat(sprintf("Rango de los datos: %.4f\n", R))

# Tamaño
n <- length(prices)
cat(sprintf("Tamaño de la variable prices: %d\n", n))

# Regla de Sturges
k <- round(1 + 3.322 * log10(n))
cat(sprintf("Regla de Sturges para la variable prices: %d\n", k))

# Cortes de intervalos
cortes <- seq(min(prices, na.rm = TRUE),
              max(prices, na.rm = TRUE),
              length.out = k + 1)

# Intervalos [a, b)
intervalos <- cut(prices,
                  breaks = cortes,
                  right = FALSE,
                  include.lowest = TRUE)

tabla <- table(intervalos)

D. Dispersión con regresión lineal por sexo
--------------------------------------------------------------------------------
import scipy.stats as stats

y_col = 'PESO'
x_col = 'ESTATURA'
custom_palette = {'Femenino': 'deeppink', 'Masculino': 'dodgerblue'}

plt.figure(figsize=(10, 7))
ax = sns.scatterplot(data=merged, x=x_col, y=y_col, hue='SEXO',
                     palette=custom_palette, s=80, alpha=0.8)

for i, (gender, color) in enumerate(custom_palette.items()):
    subset = merged[merged['SEXO'] == gender]

    pearson_corr, _ = stats.pearsonr(subset[x_col], subset[y_col])
    r_squared = pearson_corr**2
    slope, intercept, _, _, _ = stats.linregress(subset[x_col], subset[y_col])

    sns.regplot(data=subset, x=x_col, y=y_col, scatter=False,
                color=color, ax=ax, line_kws={'linewidth': 2})

    xmin, xmax = ax.get_xlim()
    ymin, ymax = ax.get_ylim()

    if gender == 'Femenino':
        text_x_pos = xmin + (xmax - xmin) * 0.05
        text_y_pos_r = ymax - (ymax - ymin) * 0.05
        text_y_pos_eq = ymax - (ymax - ymin) * 0.13
        ha = 'left'
    else:
        text_x_pos = xmax - (xmax - xmin) * 0.05
        text_y_pos_r = ymax - (ymax - ymin) * 0.05
        text_y_pos_eq = ymax - (ymax - ymin) * 0.13
        ha = 'right'

    ax.text(text_x_pos, text_y_pos_r,
            f'R ({gender}): {pearson_corr:.2f}\nR² ({gender}): {r_squared:.2f}',
            color=color, fontsize=10, weight='bold', ha=ha, va='top',
            bbox=dict(boxstyle="round,pad=0.3", fc="white", ec=color, lw=1, alpha=0.7))

    ax.text(text_x_pos, text_y_pos_eq,
            f'Y = {slope:.2f}X + {intercept:.2f}',
            color=color, fontsize=10, weight='bold', ha=ha, va='top',
            bbox=dict(boxstyle="round,pad=0.3", fc="white", ec=color, lw=1, alpha=0.7))

plt.title('Dispersión de Peso vs. Estatura por Sexo con Líneas de Regresión', fontsize=14)
